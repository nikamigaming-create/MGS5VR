cmake_minimum_required(VERSION 3.24)
project(MGS5VR VERSION 0.1.0 LANGUAGES C CXX ASM_MASM)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "MGS5VR requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(DYNAMIC_LOADER OFF CACHE BOOL "" FORCE)
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(openxr GIT_REPOSITORY https://github.com/KhronosGroup/OpenXR-SDK.git
  GIT_TAG 977f6675bc0057d5a54ed290cb5c71c699b1c0ab)
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
  GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
FetchContent_MakeAvailable(openxr minhook)
add_library(mgs5vr_core STATIC src/core.cpp src/input_bridge.cpp src/head_camera.cpp src/stereo.cpp src/arm_ik.cpp)
target_include_directories(mgs5vr_core PUBLIC include)
add_library(mgs5vr_runtime STATIC src/log.cpp src/mailbox.cpp src/xr_runtime.cpp src/scene_capture.cpp src/ui_renderer.cpp src/player_visibility.cpp src/controller_rig.cpp)
target_include_directories(mgs5vr_runtime PUBLIC include)
target_compile_definitions(mgs5vr_runtime PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX
  XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
target_link_libraries(mgs5vr_runtime PUBLIC mgs5vr_core OpenXR::openxr_loader d3d11 dxgi minhook)
add_library(dinput8 SHARED src/proxy.cpp src/capture_hook.cpp src/gamepad_hook.cpp src/camera_observer.cpp src/camera_consumer.cpp src/render_camera.cpp src/camera_bridge.asm src/process_exit.cpp src/dinput8.def)
target_link_libraries(dinput8 PRIVATE mgs5vr_runtime minhook bcrypt version)
add_executable(mgs5vr_probe src/probe.cpp)
target_link_libraries(mgs5vr_probe PRIVATE mgs5vr_runtime)
enable_testing()
add_executable(mgs5vr_tests tests/core_tests.cpp)
target_link_libraries(mgs5vr_tests PRIVATE mgs5vr_core)
add_test(NAME contracts COMMAND mgs5vr_tests)
add_executable(mgs5vr_gpu_tests tests/gpu_tests.cpp)
target_link_libraries(mgs5vr_gpu_tests PRIVATE mgs5vr_runtime)
add_test(NAME d3d11_mailbox COMMAND mgs5vr_gpu_tests)
add_executable(mgs5vr_proxy_tests tests/proxy_tests.cpp)
target_link_libraries(mgs5vr_proxy_tests PRIVATE dinput8 dxguid)
add_test(NAME system_dinput_forwarding COMMAND mgs5vr_proxy_tests)
add_executable(mgs5vr_camera_observer_tests tests/camera_observer_tests.cpp src/camera_observer.cpp src/camera_consumer.cpp src/render_camera.cpp src/camera_bridge.asm)
target_link_libraries(mgs5vr_camera_observer_tests PRIVATE mgs5vr_runtime minhook)
add_test(NAME camera_observer_passthrough COMMAND mgs5vr_camera_observer_tests)
add_executable(mgs5vr_process_exit_tests tests/process_exit_tests.cpp src/process_exit.cpp)
target_link_libraries(mgs5vr_process_exit_tests PRIVATE mgs5vr_runtime minhook)
add_test(NAME process_exit_cleanup COMMAND mgs5vr_process_exit_tests)
foreach(t mgs5vr_core mgs5vr_runtime dinput8 mgs5vr_probe mgs5vr_tests mgs5vr_gpu_tests mgs5vr_proxy_tests mgs5vr_camera_observer_tests mgs5vr_process_exit_tests)
  target_compile_options(${t} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/permissive-;/EHsc;/utf-8>")
endforeach()
install(TARGETS dinput8 mgs5vr_probe RUNTIME DESTINATION . COMPONENT MGS5VR)
install(FILES config/mgs5vr.ini README.md LICENSE CONTRIBUTING.md DESTINATION . COMPONENT MGS5VR)
install(DIRECTORY docs/ DESTINATION docs COMPONENT MGS5VR FILES_MATCHING PATTERN "*.md")
install(FILES tools/install.ps1 tools/uninstall.ps1 tools/launch-simulator.ps1 DESTINATION tools COMPONENT MGS5VR)
install(FILES profiles/tpp-1.0.15.4.json DESTINATION profiles COMPONENT MGS5VR)
install(FILES "${openxr_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME OpenXR.txt COMPONENT MGS5VR)
install(FILES "${minhook_SOURCE_DIR}/LICENSE.txt" DESTINATION licenses RENAME MinHook.txt COMPONENT MGS5VR)
install(FILES "${openxr_SOURCE_DIR}/src/external/jsoncpp/LICENSE" DESTINATION licenses RENAME JsonCpp.txt COMPONENT MGS5VR)
install(DIRECTORY "${openxr_SOURCE_DIR}/LICENSES/" DESTINATION licenses/OpenXR COMPONENT MGS5VR)
